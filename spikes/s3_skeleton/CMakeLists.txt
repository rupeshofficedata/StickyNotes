cmake_minimum_required(VERSION 3.21)
project(s3_skeleton LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets DBus Qml Quick QuickControls2 QuickDialogs2)
find_package(KF6DBusAddons REQUIRED)
find_package(KF6StatusNotifierItem REQUIRED)
add_executable(s3_skeleton main.cpp)
target_link_libraries(s3_skeleton PRIVATE Qt6::Widgets Qt6::Gui Qt6::DBus Qt6::Quick KF6::DBusAddons KF6::StatusNotifierItem)
